from pydantic import BaseModel
from typing import Optional
from enum import Enum

class ConceptClassification(str, Enum):
    CORE = "CORE"           # >85% dei syllabus

    SPECIFICO = "SPECIFICO" # <40%

class Concept(BaseModel):
    id: str
    canonical_name: str
    variants: list[str]           # Sinonimi trovati
    frequency_absolute: int       # N. syllabus che lo citano
    frequency_percentage: float
    classification: ConceptClassification
    parent_module_id: Optional[str]

class ChemistryVocabulary:
    """Gestisce il vocabolario chimico per normalizzazione e riconoscimento."""
    
    def __init__(self):
        self.synonyms = SYNONYM_MAP
        self.abbreviations = ABBREVIATIONS
        self.stopwords = ACADEMIC_STOPWORDS
        self._compile_patterns()
    
    def _compile_patterns(self):
        """Pre-compila i pattern regex per efficienza."""
        self.compiled_patterns = [
            (re.compile(pattern, re.IGNORECASE), label)
            for pattern, label in CHEMISTRY_PATTERNS
        ]
    
    def normalize(self, term: str) -> str:
        """
        Normalizza un termine al suo formato canonico.
        
        Args:
            term: Termine grezzo estratto
            
        Returns:
            Termine normalizzato (canonico)
        """
        # Lowercase e pulizia spazi
        normalized = term.lower().strip()
        normalized = re.sub(r'\s+', ' ', normalized)
        
        # Controlla sinonimi
        if normalized in self.synonyms:
            return self.synonyms[normalized]
        
        # Controlla abbreviazioni
        term_upper = term.upper().strip()
        if term_upper in self.abbreviations:
            # Ritorna l'abbreviazione stessa come forma canonica
            return term_upper
        
        return normalized
    
    def is_stopword(self, term: str) -> bool:
        """Verifica se un termine è una stopword accademica."""
        return term.lower().strip() in self.stopwords
    
    def extract_entities(self, text: str) -> list[tuple[str, str, int, int]]:
        """
        Estrae entità chimiche dal testo usando pattern matching.
        
        Args:
            text: Testo da analizzare
            
        Returns:
            Lista di tuple (match, tipo, start, end)
        """
        entities = []
        for pattern, label in self.compiled_patterns:
            for match in pattern.finditer(text):
                entities.append((
                    match.group(),
                    label,
                    match.start(),
                    match.end()
                ))
        return entities
    
    def expand_abbreviation(self, abbrev: str) -> str | None:
        """Espande un'abbreviazione al suo significato completo."""
        return self.abbreviations.get(abbrev.upper())
    
    def get_canonical_form(self, term: str) -> tuple[str, bool]:
        """
        Restituisce la forma canonica e indica se è stata trovata.
        
        Returns:
            Tuple (forma_canonica, trovata_nel_vocabolario)
        """
        normalized = self.normalize(term)
        # Se è diverso dall'input, è stato normalizzato
        is_known = normalized != term.lower().strip()
        return normalized, is_known

class Concept(BaseModel):
    """Concetto normalizzato e classificato."""
    
    id: str = Field(..., description="ID univoco del concetto")
    canonical_name: str = Field(..., description="Nome canonico normalizzato")
    
    # Varianti e fonti
    variants: list[str] = Field(

        description="Varianti testuali trovate nei syllabus"
    )

    )
    
    # Frequenza e classificazione
    frequency_absolute: int = Field(

    )

    entity_type: EntityType = EntityType.GENERIC

    embedding: Optional[list[float]] = None
    
    # Metadata
    created_at: datetime = Field(default_factory=datetime.now)
    manually_validated: bool = False
    
    class Config:
        use_enum_values = True
    
    def add_variant(self, variant: str):
        """Aggiunge una variante testuale se non già presente."""
        if variant.lower() != self.canonical_name.lower():
            if variant not in self.variants:
                self.variants.append(variant)
    
    def add_source(self, syllabus_id: str):
        """Aggiunge un syllabus sorgente."""
        if syllabus_id not in self.source_syllabus_ids:
            self.source_syllabus_ids.append(syllabus_id)
            self.frequency_absolute = len(self.source_syllabus_ids)
    
    def compute_classification(self, total_syllabus: int, 
                                threshold_core: float = 0.85,
                                threshold_comune: float = 0.40):
        """Calcola la classificazione basata sulla frequenza."""
        if total_syllabus == 0:
            return
        
        self.frequency_percentage = (self.frequency_absolute / total_syllabus) * 100
        
        if self.frequency_absolute / total_syllabus >= threshold_core:
            self.classification = ConceptClassification.CORE
        elif self.frequency_absolute / total_syllabus >= threshold_comune:
            self.classification = ConceptClassification.COMUNE
        else:
            self.classification = ConceptClassification.SPECIFICO


class ConceptCollection(BaseModel):

    total_raw_concepts_extracted: int = 0
    total_unique_concepts: int = 0
    
    # Distribuzione
    n_core: int = 0
    n_comune: int = 0
    n_specifico: int = 0
    
    created_at: datetime = Field(default_factory=datetime.now)
    
    def add_concept(self, concept: Concept):
        """Aggiunge un concetto alla collezione."""
        self.concepts.append(concept)
        self.total_unique_concepts = len(self.concepts)
    
    def compute_statistics(self):
        """Calcola statistiche sulla collezione."""
        self.n_core = sum(
            1 for c in self.concepts 
            if c.classification == ConceptClassification.CORE
        )
        self.n_comune = sum(
            1 for c in self.concepts 
            if c.classification == ConceptClassification.COMUNE
        )
        self.n_specifico = sum(
            1 for c in self.concepts 
            if c.classification == ConceptClassification.SPECIFICO
        )
    
    def get_by_classification(
        self, 
        classification: ConceptClassification
    ) -> list[Concept]:
        """Restituisce concetti filtrati per classificazione."""
        return [c for c in self.concepts if c.classification == classification]
    
    def to_frequency_dict(self) -> dict[str, int]:
        """Restituisce dizionario nome ? frequenza."""
        return {c.canonical_name: c.frequency_absolute for c in self.concepts}

class ConceptExtractor:
    """
    Estrattore di concetti chimici da testi di syllabus.
    
    Pipeline:
    1. Preprocessing testo
    2. NLP (spaCy) per tokenizzazione, lemmatizzazione, POS
    3. Pattern matching per entità chimiche
    4. Estrazione n-grammi rilevanti
    5. Normalizzazione via vocabolario
    6. Deduplicazione e aggregazione
    """
    
    def __init__(self, vocabulary: ChemistryVocabulary | None = None):
        """
        Inizializza l'estrattore.
        
        Args:
            vocabulary: Vocabolario chimico per normalizzazione.
                       Se None, ne crea uno di default.
        """
        self.vocab = vocabulary or ChemistryVocabulary()
        self._load_spacy_model()
        
        # Cache per concetti già visti (nome_canonico ? Concept)
        self._concept_cache: dict[str, Concept] = {}
        
        # Contatore per ID univoci
        self._id_counter = 0
    
    def _load_spacy_model(self):
        """Carica il modello spaCy per l'italiano."""
        try:
            self.nlp = spacy.load(SPACY_MODEL)
        except OSError:
            print(f"Modello {SPACY_MODEL} non trovato. Scaricamento...")
            spacy.cli.download(SPACY_MODEL)
            self.nlp = spacy.load(SPACY_MODEL)
        
        # Aggiungi pipe custom per chimica se necessario
        # (per ora usiamo pattern matching manuale)
    
    def _generate_id(self, text: str) -> str:
        """Genera ID univoco basato su hash del testo."""
        self._id_counter += 1
        hash_part = hashlib.md5(text.encode()).hexdigest()[:8]
        return f"concept_{self._id_counter:05d}_{hash_part}"
    
    def preprocess_text(self, text: str) -> str:
        """
        Pulisce e normalizza il testo grezzo.
        
        Args:
            text: Testo grezzo (possibilmente da OCR/PDF)
            
        Returns:
            Testo pulito
        """
        # Rimuovi caratteri di controllo e normalizza spazi
        text = re.sub(r'[\x00-\x08\x0b\x0c\x0e-\x1f\x7f-\x9f]', '', text)
        text = re.sub(r'\s+', ' ', text)
        
        # Normalizza trattini e apostrofi
        text = text.replace('–', '-').replace('—', '-')
        text = text.replace(''', "'").replace(''', "'")
        
        # Rimuovi numeri di pagina isolati
        text = re.sub(r'\n\s*\d+\s*\n', '\n', text)
        
        # Rimuovi header/footer comuni nei PDF accademici
        text = re.sub(
            r'(Pag\.\s*\d+\s*/\s*\d+|Università\s+degli\s+Studi.*?\n)', 
            '', 
            text, 
            flags=re.IGNORECASE
        )
        
        return text.strip()
    
    def extract_program_section(self, text: str) -> str:
        """
        Estrae la sezione del programma/contenuti dal syllabus.
        Ignora sezioni come prerequisiti, modalità esame, ecc.
        
        Args:
            text: Testo completo del syllabus
            
        Returns:
            Solo la sezione rilevante (programma/contenuti)
        """
        # Pattern per identificare inizio sezione programma
        program_patterns = [
            r'(?:PROGRAMMA|CONTENUTI|SYLLABUS|PROGRAMMA.SYLLABUS)[:\s]*\n',
            r'(?:Contenuti|Programma)\s*(?:del\s+corso|dell.insegnamento)?[:\s]*\n',
            r'(?:PROGRAMMA.ESTESO|Programma\s+esteso)[:\s]*\n',
        ]
        
        # Pattern per fine sezione programma
        end_patterns = [
            r'\n(?:MATERIALE\s+DIDATTICO|TESTI|BIBLIOGRAFIA)',
            r'\n(?:MODALITÀ\s+DI\s+(?:VERIFICA|ESAME|VALUTAZIONE))',
            r'\n(?:METODI\s+DIDATTICI|METODOLOGIE)',
            r'\n(?:RISULTATI\s+DI\s+APPRENDIMENTO)',
            r'\n(?:Testi?\s+(?:consigliati?|di\s+riferimento))',
        ]
        
        # Trova inizio
        start_pos = 0
        for pattern in program_patterns:
            match = re.search(pattern, text, re.IGNORECASE)
            if match:
                start_pos = match.end()
                break
        
        # Trova fine
        end_pos = len(text)
        remaining_text = text[start_pos:]
        for pattern in end_patterns:
            match = re.search(pattern, remaining_text, re.IGNORECASE)
            if match:
                end_pos = start_pos + match.start()
                break
        
        return text[start_pos:end_pos].strip()
    
    def extract_raw_concepts(
        self, 
        text: str, 
        syllabus_id: str
    ) -> list[RawConcept]:
        """
        Estrae concetti grezzi dal testo.
        
        Args:
            text: Testo (preferibilmente già filtrato alla sezione programma)
            syllabus_id: ID del syllabus sorgente
            
        Returns:
            Lista di concetti grezzi
        """
        raw_concepts = []
        
        # 1. Pattern matching per entità chimiche note
        entities = self.vocab.extract_entities(text)
        for match_text, entity_type, start, end in entities:
            # Estrai contesto (frase contenente il match)
            context_start = max(0, text.rfind('.', 0, start) + 1)
            context_end = text.find('.', end)
            if context_end == -1:
                context_end = min(len(text), end + 100)
            context = text[context_start:context_end].strip()
            
            raw_concepts.append(RawConcept(
                text=match_text,
                source_syllabus_id=syllabus_id,
                position_in_text=start,
                context=context,
                entity_type=EntityType(entity_type) if entity_type in EntityType.__members__ else EntityType.GENERIC,
                confidence=0.9  # Alta confidenza per pattern noti
            ))
        
        # 2. NLP per estrazione basata su POS tagging
        doc = self.nlp(text)
        nlp_concepts = self._extract_from_doc(doc, syllabus_id, text)
        raw_concepts.extend(nlp_concepts)
        
        # 3. N-grammi rilevanti
        ngram_concepts = self._extract_ngrams(doc, syllabus_id, text)
        raw_concepts.extend(ngram_concepts)
        
        return raw_concepts
    
    def _extract_from_doc(
        self, 
        doc: Doc, 
        syllabus_id: str,
        original_text: str
    ) -> list[RawConcept]:
        """Estrae concetti usando analisi NLP."""
        concepts = []
        
        # Estrai noun chunks (sintagmi nominali)
        for chunk in doc.noun_chunks:
            # Filtra chunk troppo corti o stopword
            chunk_text = chunk.text.strip()
            if len(chunk_text) < MIN_CONCEPT_LENGTH:
                continue
            if self.vocab.is_stopword(chunk_text):
                continue
            
            # Verifica se contiene termini chimici rilevanti
            if self._is_chemistry_relevant(chunk):
                context = self._get_sentence_context(chunk, original_text)
                concepts.append(RawConcept(
                    text=chunk_text,
                    source_syllabus_id=syllabus_id,
                    position_in_text=chunk.start_char,
                    context=context,
                    entity_type=EntityType.GENERIC,
                    confidence=0.7
                ))
        
        return concepts
    
    def _extract_ngrams(
        self, 
        doc: Doc, 
        syllabus_id: str,
        original_text: str
    ) -> list[RawConcept]:
        """Estrae n-grammi rilevanti."""
        concepts = []
        tokens = [t for t in doc if not t.is_punct and not t.is_space]
        
        for n in range(2, MAX_NGRAM_SIZE + 1):
            for i in range(len(tokens) - n + 1):
                ngram_tokens = tokens[i:i+n]
                ngram_text = ' '.join(t.text for t in ngram_tokens)
                
                # Verifica se è un termine chimico noto
                normalized, is_known = self.vocab.get_canonical_form(ngram_text)
                
                if is_known:
                    start_char = ngram_tokens[0].idx
                    context = self._get_sentence_context(
                        ngram_tokens[0], 
                        original_text
                    )
                    concepts.append(RawConcept(
                        text=ngram_text,
                        source_syllabus_id=syllabus_id,
                        position_in_text=start_char,
                        context=context,
                        entity_type=EntityType.GENERIC,
                        confidence=0.85
                    ))
        
        return concepts
    
    def _is_chemistry_relevant(self, span: Span) -> bool:
        """Verifica se uno span contiene termini chimicamente rilevanti."""
        text_lower = span.text.lower()
        
        # Controlla se è nel vocabolario
        _, is_known = self.vocab.get_canonical_form(text_lower)
        if is_known:
            return True
        
        # Controlla parole chiave chimiche
        chemistry_keywords = {
            'reazione', 'sintesi', 'gruppo', 'legame', 'molecola',
            'composto', 'acido', 'base', 'ossidazione', 'riduzione',
            'addizione', 'eliminazione', 'sostituzione', 'meccanismo',
            'stereo', 'isomer', 'chiral', 'enantio', 'diastereo',
            'carbonio', 'ossigeno', 'azoto', 'idrogeno', 'alogeno',
            'alchil', 'aril', 'acil', 'carbonil', 'carbossil',
            'ammino', 'ammid', 'ester', 'eter', 'alcol', 'fenol',
            'zuccher', 'glucos', 'fruttosi', 'saccarid', 'glicos',
            'lipid', 'protein', 'peptid', 'nucleotid', 'nucleosid'
        }
        
        for keyword in chemistry_keywords:
            if keyword in text_lower:
                return True
        
        return False
    
    def _get_sentence_context(self, token_or_span, text: str) -> str:
        """Estrae la frase di contesto per un token o span."""
        if hasattr(token_or_span, 'sent'):
            return token_or_span.sent.text.strip()
        
        # Fallback: estrai ~100 caratteri intorno
        if hasattr(token_or_span, 'idx'):
            pos = token_or_span.idx
        elif hasattr(token_or_span, 'start_char'):
            pos = token_or_span.start_char
        else:
            return ""
        
        start = max(0, pos - 50)
        end = min(len(text), pos + 100)
        return text[start:end].strip()
    
    def normalize_concepts(
        self, 
        raw_concepts: list[RawConcept]
    ) -> dict[str, list[RawConcept]]:
        """
        Normalizza e raggruppa concetti grezzi.
        
        Args:
            raw_concepts: Lista di concetti grezzi
            
        Returns:
            Dizionario nome_canonico ? lista di RawConcept raggruppati
        """
        grouped: dict[str, list[RawConcept]] = defaultdict(list)
        
        for raw in raw_concepts:
            canonical = self.vocab.normalize(raw.text)
            
            # Salta se troppo corto dopo normalizzazione
            if len(canonical) < MIN_CONCEPT_LENGTH:
                continue
            
            # Salta stopword accademiche
            if self.vocab.is_stopword(canonical):
                continue
            
            grouped[canonical].append(raw)
        
        return dict(grouped)
    
    def build_concept_collection(
        self,
        grouped_concepts: dict[str, list[RawConcept]],
        total_syllabus: int,
        collection_name: str = "Chimica Organica L-13"
    ) -> ConceptCollection:
        """
        Costruisce la collezione finale di concetti normalizzati.
        
        Args:
            grouped_concepts: Dizionario da normalize_concepts()
            total_syllabus: Numero totale di syllabus analizzati
            collection_name: Nome della collezione
            
        Returns:
            ConceptCollection con tutti i concetti classificati
        """
        collection = ConceptCollection(
            id=self._generate_id(collection_name),
            name=collection_name,
            total_syllabus_analyzed=total_syllabus
        )
        
        for canonical_name, raw_list in grouped_concepts.items():
            # Determina tipo entità più comune
            entity_types = [r.entity_type for r in raw_list]
            most_common_type = max(set(entity_types), key=entity_types.count)
            
            # Raccogli varianti uniche
            variants = list(set(
                r.text for r in raw_list 
                if r.text.lower() != canonical_name.lower()
            ))
            
            # Raccogli syllabus sorgenti unici
            source_ids = list(set(r.source_syllabus_id for r in raw_list))
            
            concept = Concept(
                id=self._generate_id(canonical_name),
                canonical_name=canonical_name,
                variants=variants,
                source_syllabus_ids=source_ids,
                frequency_absolute=len(source_ids),
                entity_type=most_common_type
            )
            
            # Calcola classificazione
            concept.compute_classification(
                total_syllabus,
                threshold_core=THRESHOLD_CORE,
                threshold_comune=THRESHOLD_COMUNE
            )
            
            collection.add_concept(concept)
        
        # Calcola statistiche finali
        collection.compute_statistics()
        collection.total_raw_concepts_extracted = sum(
            len(v) for v in grouped_concepts.values()
        )
        
        return collection
    
    def extract_from_syllabus(
        self,
        text: str,
        syllabus_id: str,
        extract_program_only: bool = True
    ) -> list[RawConcept]:
        """
        Pipeline completa di estrazione da un singolo syllabus.
        
        Args:
            text: Testo completo del syllabus
            syllabus_id: ID univoco del syllabus
            extract_program_only: Se True, estrae solo dalla sezione programma
            
        Returns:
            Lista di concetti grezzi estratti
        """
        # Preprocessing
        cleaned_text = self.preprocess_text(text)
        
        # Estrai solo sezione programma se richiesto
        if extract_program_only:
            program_text = self.extract_program_section(cleaned_text)
            # Fallback: usa tutto il testo se la sezione non è trovata
            if len(program_text) < 100:
                program_text = cleaned_text
        else:
            program_text = cleaned_text
        
        # Estrai concetti grezzi
        raw_concepts = self.extract_raw_concepts(program_text, syllabus_id)
        
        return raw_concepts
    
    def process_multiple_syllabus(
        self,
        syllabus_texts: dict[str, str],
        collection_name: str = "Chimica Organica L-13"
    ) -> ConceptCollection:
        """
        Processa multipli syllabus e costruisce la collezione.
        
        Args:
            syllabus_texts: Dizionario syllabus_id ? testo
            collection_name: Nome per la collezione risultante
            
        Returns:
            ConceptCollection con concetti aggregati e classificati
        """
        all_raw_concepts: list[RawConcept] = []
        
        for syllabus_id, text in syllabus_texts.items():
            raw = self.extract_from_syllabus(text, syllabus_id)
            all_raw_concepts.extend(raw)
        
        # Normalizza e raggruppa
        grouped = self.normalize_concepts(all_raw_concepts)
        
        # Costruisci collezione
        collection = self.build_concept_collection(
            grouped,
            total_syllabus=len(syllabus_texts),
            collection_name=collection_name
        )
        
        return collection


# Funzione di utilità per uso rapido
def extract_concepts_from_texts(
    texts: dict[str, str],
    collection_name: str = "Chimica Organica L-13"
) -> ConceptCollection:
    """

    Args:
        texts: Dizionario id ? testo dei syllabus
        collection_name: Nome collezione
        
    Returns:
        ConceptCollection
    """
    extractor = ConceptExtractor()
    return extractor.process_multiple_syllabus(texts, collection_name)

embeddings.py
Gestione embeddings per similarità semantica tra concetti
"""

import numpy as np
from typing import Optional
from functools import lru_cache

try:
    from sentence_transformers import SentenceTransformer
    HAS_SENTENCE_TRANSFORMERS = True
except ImportError:
    HAS_SENTENCE_TRANSFORMERS = False
    print("Warning: sentence-transformers non installato. Usando fallback.")


class EmbeddingModel:
    """
    Wrapper per modelli di embedding semantico.
    Supporta sentence-transformers con fallback a TF-IDF.
    """
    
    # Modelli consigliati per italiano/multilingua
    RECOMMENDED_MODELS = [
        "paraphrase-multilingual-MiniLM-L12-v2",  # Buon bilanciamento
        "distiluse-base-multilingual-cased-v1",   # Leggero
        "paraphrase-multilingual-mpnet-base-v2",  # Più accurato ma pesante
    ]
    
    def __init__(self, model_name: Optional[str] = None):
        """
        Inizializza il modello di embedding.
        
        Args:
            model_name: Nome modello sentence-transformers. 
                       Se None, usa il primo disponibile.
        """
        self.model_name = model_name or self.RECOMMENDED_MODELS[0]
        self.model = None
        self.dimension = None
        self._fallback_mode = False
        
        self._load_model()
    
    def _load_model(self):
        """Carica il modello di embedding."""
        if HAS_SENTENCE_TRANSFORMERS:
            try:
                print(f"Caricamento modello embedding: {self.model_name}")
                self.model = SentenceTransformer(self.model_name)
                # Determina dimensione con una query di test
                test_emb = self.model.encode(["test"])
                self.dimension = test_emb.shape[1]
                print(f"Modello caricato. Dimensione embedding: {self.dimension}")
            except Exception as e:
                print(f"Errore caricamento modello: {e}. Uso fallback TF-IDF.")
                self._fallback_mode = True
        else:
            self._fallback_mode = True
        
        if self._fallback_mode:
            self._init_fallback()
    
    def _init_fallback(self):
        """Inizializza modalità fallback con TF-IDF."""
        from sklearn.feature_extraction.text import TfidfVectorizer
        self.vectorizer = TfidfVectorizer(
            max_features=500,
            ngram_range=(1, 2),
            lowercase=True
        )
        self.dimension = 500  # Sarà aggiustato dopo fit
        self._fitted = False
    
    def encode(self, texts: list[str]) -> np.ndarray:
        """
        Genera embeddings per una lista di testi.
        
        Args:
            texts: Lista di stringhe da codificare
            
        Returns:
            Array numpy di shape (n_texts, dimension)
        """
        if not texts:
            return np.array([])
        
        if self._fallback_mode:
            return self._encode_tfidf(texts)
        
        # Sentence-transformers
        embeddings = self.model.encode(
            texts,
            show_progress_bar=len(texts) > 100,
            convert_to_numpy=True
        )
        return embeddings
    
    def _encode_tfidf(self, texts: list[str]) -> np.ndarray:
        """Encoding fallback con TF-IDF."""
        if not self._fitted:
            embeddings = self.vectorizer.fit_transform(texts).toarray()
            self._fitted = True
            self.dimension = embeddings.shape[1]
        else:
            embeddings = self.vectorizer.transform(texts).toarray()
        return embeddings
    
    def similarity(self, emb1: np.ndarray, emb2: np.ndarray) -> float:
        """
        Calcola similarità coseno tra due embeddings.
        
        Args:
            emb1, emb2: Vettori embedding
            
        Returns:
            Similarità coseno (0-1)
        """
        norm1 = np.linalg.norm(emb1)
        norm2 = np.linalg.norm(emb2)
        
        if norm1 == 0 or norm2 == 0:
            return 0.0
        
        return float(np.dot(emb1, emb2) / (norm1 * norm2))
    
    def pairwise_similarity_matrix(
        self, 
        embeddings: np.ndarray
    ) -> np.ndarray:
        """
        Calcola matrice di similarità tra tutti i vettori.
        
        Args:
            embeddings: Array (n, dim)
            
        Returns:
            Matrice (n, n) di similarità
        """
        # Normalizza
        norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
        norms[norms == 0] = 1  # Evita divisione per zero
        normalized = embeddings / norms
        
        # Prodotto matriciale per similarità coseno
        similarity_matrix = np.dot(normalized, normalized.T)
        
        return similarity_matrix


class ConceptEmbedder:
    """
    Specializzato per embedding di concetti chimici.
    Arricchisce il testo con contesto prima dell'encoding.
    """
    
    def __init__(self, embedding_model: Optional[EmbeddingModel] = None):
        """
        Args:
            embedding_model: Modello di embedding. Se None, ne crea uno.
        """
        self.model = embedding_model or EmbeddingModel()
        self._cache: dict[str, np.ndarray] = {}
    
    def _enrich_concept_text(self, concept_name: str, variants: list[str] = None) -> str:
        """
        Arricchisce il nome del concetto con varianti per embedding migliore.
        
        Args:
            concept_name: Nome canonico del concetto
            variants: Varianti testuali
            
        Returns:
            Testo arricchito per embedding
        """
        parts = [concept_name]
        
        if variants:
            # Aggiungi varianti significative (non troppo simili)
            for v in variants[:3]:  # Max 3 varianti
                if len(v) > 3 and v.lower() != concept_name.lower():
                    parts.append(v)
        
        return " | ".join(parts)
    
    def embed_concepts(
        self, 
        concepts: list,  # list[Concept]
        use_variants: bool = True
    ) -> dict[str, np.ndarray]:
        """
        Genera embeddings per una lista di concetti.
        
        Args:
            concepts: Lista di oggetti Concept
            use_variants: Se includere varianti nel testo
            
        Returns:
            Dizionario concept_id ? embedding
        """
        # Prepara testi
        texts = []
        ids = []
        
        for concept in concepts:
            # Controlla cache
            if concept.id in self._cache:
                continue
            
            if use_variants:
                text = self._enrich_concept_text(
                    concept.canonical_name, 
                    concept.variants
                )
            else:
                text = concept.canonical_name
            
            texts.append(text)
            ids.append(concept.id)
        
        # Genera embeddings per quelli non in cache
        if texts:
            embeddings = self.model.encode(texts)
            for i, concept_id in enumerate(ids):
                self._cache[concept_id] = embeddings[i]
        
        # Ritorna tutti (da cache)
        return {c.id: self._cache.get(c.id) for c in concepts if c.id in self._cache}
    
    def get_similarity_matrix(
        self, 
        concepts: list
    ) -> tuple[np.ndarray, list[str]]:
        """
        Costruisce matrice di similarità tra concetti.
        
        Args:
            concepts: Lista di Concept
            
        Returns:
            Tuple (matrice_similarità, lista_id_concetti)
        """
        # Assicura embeddings
        embeddings_dict = self.embed_concepts(concepts)
        
        # Ordina per ID consistente
        sorted_concepts = sorted(concepts, key=lambda c: c.id)
        ids = [c.id for c in sorted_concepts]
        
        # Costruisci matrice embeddings
        embedding_matrix = np.array([
            embeddings_dict[c.id] for c in sorted_concepts
        ])
        
        # Calcola similarità
        sim_matrix = self.model.pairwise_similarity_matrix(embedding_matrix)
        
        return sim_matrix, ids

import numpy as np
from collections import defaultdict

class CooccurrenceAnalyzer:
    """
    Analizza la co-occorrenza di concetti nei syllabus.
    Due concetti co-occorrono se appaiono nello stesso syllabus.
    """
    
    def __init__(self):
        self._matrix: Optional[np.ndarray] = None
        self._concept_ids: list[str] = []
        self._id_to_index: dict[str, int] = {}
    
    def build_cooccurrence_matrix(
        self, 
        collection: ConceptCollection
    ) -> np.ndarray:
        """
        Costruisce matrice di co-occorrenza.
        
        Args:
            collection: Collezione di concetti
            
        Returns:
            Matrice (n_concepts, n_concepts) dove M[i,j] = 
            numero di syllabus in cui i e j appaiono insieme
        """
        concepts = collection.concepts
        n = len(concepts)
        
        # Mappa ID ? indice
        self._concept_ids = [c.id for c in concepts]
        self._id_to_index = {cid: i for i, cid in enumerate(self._concept_ids)}
        
        # Inizializza matrice
        self._matrix = np.zeros((n, n), dtype=np.float32)
        
        # Costruisci indice inverso: syllabus_id ? set di concept_ids
        syllabus_to_concepts: dict[str, set[str]] = defaultdict(set)
        
        for concept in concepts:
            for syllabus_id in concept.source_syllabus_ids:
                syllabus_to_concepts[syllabus_id].add(concept.id)
        
        # Conta co-occorrenze
        for syllabus_id, concept_ids in syllabus_to_concepts.items():
            concept_list = list(concept_ids)
            for i, cid1 in enumerate(concept_list):
                idx1 = self._id_to_index[cid1]
                for cid2 in concept_list[i:]:  # Include diagonale
                    idx2 = self._id_to_index[cid2]
                    self._matrix[idx1, idx2] += 1
                    if idx1 != idx2:
                        self._matrix[idx2, idx1] += 1  # Simmetrica
        
        return self._matrix
    
    def get_normalized_matrix(self) -> np.ndarray:
        """
        Restituisce matrice normalizzata (Jaccard-like).
        
        M_norm[i,j] = co-occurrence(i,j) / (freq(i) + freq(j) - co-occurrence(i,j))
        
        Returns:
            Matrice normalizzata (0-1)
        """
        if self._matrix is None:
            raise ValueError("Matrice non costruita. Chiama build_cooccurrence_matrix prima.")
        
        n = self._matrix.shape[0]
        normalized = np.zeros_like(self._matrix)
        
        # Diagonale = frequenza del concetto
        diag = np.diag(self._matrix)
        
        for i in range(n):
            for j in range(i, n):
                co_occ = self._matrix[i, j]
                union = diag[i] + diag[j] - co_occ
                
                if union > 0:
                    normalized[i, j] = co_occ / union
                    normalized[j, i] = normalized[i, j]
        
        return normalized
    
    def get_cooccurrence(self, concept_id1: str, concept_id2: str) -> int:
        """Restituisce co-occorrenza tra due concetti specifici."""
        if self._matrix is None:
            return 0
        
        idx1 = self._id_to_index.get(concept_id1)
        idx2 = self._id_to_index.get(concept_id2)
        
        if idx1 is None or idx2 is None:
            return 0
        
        return int(self._matrix[idx1, idx2])
    
    def get_top_cooccurrences(
        self, 
        concept_id: str, 
        top_n: int = 10
    ) -> list[tuple[str, int]]:
        """
        Trova i concetti che co-occorrono di più con uno dato.
        
        Args:
            concept_id: ID del concetto query
            top_n: Numero di risultati
            
        Returns:
            Lista di tuple (concept_id, conteggio)
        """
        if self._matrix is None or concept_id not in self._id_to_index:
            return []
        
        idx = self._id_to_index[concept_id]
        row = self._matrix[idx, :]
        
        # Escludi se stesso
        row_copy = row.copy()
        row_copy[idx] = -1
        
        # Ordina per co-occorrenza decrescente
        top_indices = np.argsort(row_copy)[::-1][:top_n]
        
        results = []
        for i in top_indices:
            if row_copy[i] > 0:
                results.append((self._concept_ids[i], int(row[i])))
        
        return results
    
    def get_concept_ids(self) -> list[str]:
        """Restituisce lista ordinata di concept IDs."""
        return self._concept_ids.copy()


def build_combined_similarity(
    semantic_matrix: np.ndarray,
    cooccurrence_matrix: np.ndarray,
    semantic_weight: float = 0.6,
    cooccurrence_weight: float = 0.4
) -> np.ndarray:
    """
    Combina similarità semantica e co-occorrenza.
    
    Args:
        semantic_matrix: Matrice similarità coseno (0-1)
        cooccurrence_matrix: Matrice co-occorrenza normalizzata (0-1)
        semantic_weight: Peso per similarità semantica
        cooccurrence_weight: Peso per co-occorrenza
        
    Returns:
        Matrice combinata
    """
    assert semantic_matrix.shape == cooccurrence_matrix.shape, \
        "Le matrici devono avere la stessa dimensione"
    
    assert abs(semantic_weight + cooccurrence_weight - 1.0) < 0.001, \
        "I pesi devono sommare a 1"
    
    combined = (
        semantic_weight * semantic_matrix + 
        cooccurrence_weight * cooccurrence_matrix
    )
    
    return combined

        # Bonus per moduli CORE
        if self.classification == ModuleClassification.CORE:
            base_weight *= 1.2

            base_weight *= 0.8
        
        # Normalizza tra 0.02 e 0.15
        self.suggested_weight = max(0.02, min(0.15, base_weight))

    ):
        """
        Args:
            embedding_model: Modello per embeddings. Se None, ne crea uno.
            llm_client: Client LLM per labeling. Se None, usa singleton.
            semantic_weight: Peso similarità semantica (0-1)
            cooccurrence_weight: Peso co-occorrenza (0-1)
        """
        self.embedder = ConceptEmbedder(embedding_model)
        self.cooccurrence = CooccurrenceAnalyzer()
        self.llm = llm_client or get_llm_client()
        
        self.semantic_weight = semantic_weight
        self.cooccurrence_weight = cooccurrence_weight
        
        # Cache risultati
        self._similarity_matrix: Optional[np.ndarray] = None
        self._concept_ids: list[str] = []
        self._linkage_matrix: Optional[np.ndarray] = None
    
    def _generate_id(self, prefix: str, text: str) -> str:
        """Genera ID univoco."""
        hash_part = hashlib.md5(text.encode()).hexdigest()[:8]
        return f"{prefix}_{hash_part}"
    
    def build_similarity_matrix(
        self, 
        collection: ConceptCollection
    ) -> np.ndarray:
        """
        Costruisce matrice di similarità combinata.
        
        Args:
            collection: Collezione di concetti
            
        Returns:
            Matrice di similarità (n_concepts, n_concepts)
        """
        concepts = collection.concepts
        
        print(f"Costruzione matrice similarità per {len(concepts)} concetti...")
        
        # 1. Similarità semantica
        print("  - Calcolo embeddings...")
        semantic_matrix, concept_ids = self.embedder.get_similarity_matrix(concepts)
        self._concept_ids = concept_ids
        
        # 2. Co-occorrenza
        print("  - Calcolo co-occorrenza...")
        self.cooccurrence.build_cooccurrence_matrix(collection)
        cooccurrence_matrix = self.cooccurrence.get_normalized_matrix()
        
        # Riordina cooccurrence per matchare ordine semantic
        # (entrambi usano concept.id, ma potrebbero essere ordinati diversamente)
        cooccurrence_ids = self.cooccurrence.get_concept_ids()
        
        if cooccurrence_ids != concept_ids:
            # Riordina
            id_to_idx_coocc = {cid: i for i, cid in enumerate(cooccurrence_ids)}
            reorder_indices = [id_to_idx_coocc[cid] for cid in concept_ids]
            cooccurrence_matrix = cooccurrence_matrix[np.ix_(reorder_indices, reorder_indices)]
        
        # 3. Combina
        print("  - Combinazione matrici...")
        self._similarity_matrix = build_combined_similarity(
            semantic_matrix,
            cooccurrence_matrix,
            self.semantic_weight,
            self.cooccurrence_weight
        )
        
        print(f"  - Matrice costruita: {self._similarity_matrix.shape}")
        
        return self._similarity_matrix
    
    def perform_clustering(
        self,
        n_clusters: Optional[int] = None,
        method: str = 'ward'
    ) -> np.ndarray:
        """
        Esegue clustering gerarchico.
        
        Args:
            n_clusters: Numero di cluster desiderati. Se None, usa TARGET_N_MODULES.
            method: Metodo di linkage ('ward', 'average', 'complete', 'single')
            
        Returns:
            Array di label cluster per ogni concetto
        """
        if self._similarity_matrix is None:
            raise ValueError("Costruisci prima la matrice di similarità")
        
        n_clusters = n_clusters or TARGET_N_MODULES
        
        print(f"Clustering gerarchico (target: {n_clusters} cluster)...")
        
        # Converti similarità in distanza
        distance_matrix = 1 - self._similarity_matrix
        np.fill_diagonal(distance_matrix, 0)
        
        # Assicura simmetria e non-negatività
        distance_matrix = (distance_matrix + distance_matrix.T) / 2
        distance_matrix = np.maximum(distance_matrix, 0)
        
        # Converti in forma condensata per linkage
        condensed_dist = squareform(distance_matrix, checks=False)
        
        # Linkage gerarchico
        self._linkage_matrix = linkage(condensed_dist, method=method)
        
        # Taglia dendrogramma per ottenere n_clusters
        labels = fcluster(self._linkage_matrix, n_clusters, criterion='maxclust')
        
        print(f"  - Cluster generati: {len(set(labels))}")
        
        return labels
    
    def evaluate_clustering(self, labels: np.ndarray) -> dict:
        """
        Valuta qualità del clustering.
        
        Args:
            labels: Array di label cluster
            
        Returns:
            Dict con metriche di valutazione
        """
        if self._similarity_matrix is None:
            return {}
        
        # Converti similarità in distanza per silhouette
        distance_matrix = 1 - self._similarity_matrix
        
        n_clusters = len(set(labels))
        
        metrics = {
            "n_clusters": n_clusters,
            "cluster_sizes": {},
        }
        
        # Dimensioni cluster
        for label in set(labels):
            metrics["cluster_sizes"][int(label)] = int(np.sum(labels == label))
        
        # Silhouette score (se più di 1 cluster)
        if n_clusters > 1 and n_clusters < len(labels):
            try:
                avg_silhouette = silhouette_score(distance_matrix, labels, metric='precomputed')
                sample_silhouettes = silhouette_samples(distance_matrix, labels, metric='precomputed')
                
                metrics["avg_silhouette"] = float(avg_silhouette)
                metrics["silhouette_per_cluster"] = {}
                
                for label in set(labels):
                    mask = labels == label
                    cluster_silhouette = float(np.mean(sample_silhouettes[mask]))
                    metrics["silhouette_per_cluster"][int(label)] = cluster_silhouette
                    
            except Exception as e:
                print(f"Warning: impossibile calcolare silhouette: {e}")
                metrics["avg_silhouette"] = 0.0
        
        return metrics
    
    def find_optimal_n_clusters(
        self,
        min_clusters: int = 8,
        max_clusters: int = 20
    ) -> int:
        """
        Trova numero ottimale di cluster usando silhouette score.
        
        Args:
            min_clusters: Minimo numero di cluster
            max_clusters: Massimo numero di cluster
            
        Returns:
            Numero ottimale di cluster
        """
        if self._similarity_matrix is None:
            return TARGET_N_MODULES
        
        print(f"Ricerca numero ottimale cluster ({min_clusters}-{max_clusters})...")
        
        best_score = -1
        best_n = TARGET_N_MODULES
        
        distance_matrix = 1 - self._similarity_matrix
        condensed_dist = squareform(distance_matrix, checks=False)
        linkage_matrix = linkage(condensed_dist, method='ward')
        
        for n in range(min_clusters, max_clusters + 1):
            labels = fcluster(linkage_matrix, n, criterion='maxclust')
            
            # Verifica che tutti i cluster abbiano dimensione minima
            sizes = [np.sum(labels == l) for l in set(labels)]
            if min(sizes) < MIN_CLUSTER_SIZE:
                continue
            
            try:
                score = silhouette_score(distance_matrix, labels, metric='precomputed')
                print(f"  - n={n}: silhouette={score:.3f}")
                
                if score > best_score:
                    best_score = score
                    best_n = n
            except:
                continue
        
        print(f"  - Ottimale: {best_n} cluster (silhouette={best_score:.3f})")
        
        return best_n
    
    def build_modules(
        self,
        collection: ConceptCollection,
        labels: np.ndarray,
        use_llm_labeling: bool = True
    ) -> list[Module]:
        """
        Costruisce moduli a partire dai cluster.
        
        Args:
            collection: Collezione di concetti
            labels: Array di label cluster
            use_llm_labeling: Se usare LLM per generare nomi moduli
            
        Returns:
            Lista di Module
        """
        print("Costruzione moduli da cluster...")
        
        # Mappa concept_id ? Concept
        concepts_by_id = {c.id: c for c in collection.concepts}
        
        # Raggruppa concetti per cluster
        clusters: dict[int, list[Concept]] = defaultdict(list)
        
        for i, concept_id in enumerate(self._concept_ids):
            cluster_label = int(labels[i])
            concept = concepts_by_id.get(concept_id)
            if concept:
                clusters[cluster_label].append(concept)
        
        modules = []
        
        for cluster_id, concepts in sorted(clusters.items()):
            # Ordina concetti per frequenza
            sorted_concepts = sorted(
                concepts, 
                key=lambda c: c.frequency_percentage, 
                reverse=True
            )
            
            # Genera nome modulo
            concept_names = [c.canonical_name for c in sorted_concepts[:10]]
            
            if use_llm_labeling and self.llm.is_available():
                module_name, description = self.llm.label_cluster(concept_names)
            else:
                # Fallback: usa concetto più frequente
                module_name = self._generate_fallback_name(concept_names)
                description = f"Modulo contenente: {', '.join(concept_names[:5])}"
            
            module = Module(
                id=self._generate_id("mod", module_name),
                name=module_name,
                description=description,
                cluster_id=cluster_id
            )
            
            # Aggiungi concetti
            for concept in sorted_concepts:
                module.add_concept(concept)
            
            # Calcola peso suggerito
            module.compute_suggested_weight(collection.total_unique_concepts)
            
            modules.append(module)
        
        print(f"  - Generati {len(modules)} moduli")
        
        return modules

    def order_modules(
        self, 
        modules: list[Module],
        use_llm: bool = True
    ) -> list[Module]:
        """
        Ordina moduli in sequenza didattica logica.
        
        Args:
            modules: Lista di moduli
            use_llm: Se usare LLM per ordinamento
            
        Returns:
            Lista ordinata di moduli
        """
        print("Ordinamento moduli...")
        
        if use_llm and self.llm.is_available():
            module_names = [m.name for m in modules]
            ordered_info = self.llm.suggest_module_hierarchy(module_names)
            
            # Mappa nome ? info ordinamento
            order_map = {info["name"]: info for info in ordered_info}
            
            for module in modules:
                if module.name in order_map:
                    module.order = order_map[module.name].get("order", 0)
                    module.group = order_map[module.name].get("group", "")
        else:
            # Fallback: ordina per classificazione e frequenza
            def sort_key(m):
                class_order = {
                    ModuleClassification.CORE: 0,
                    ModuleClassification.COMUNE: 1,
                    ModuleClassification.SPECIFICO: 2
                }
                return (class_order.get(m.classification, 1), -m.avg_frequency)
            
            modules_sorted = sorted(modules, key=sort_key)
            for i, module in enumerate(modules_sorted):
                module.order = i + 1
            
            modules = modules_sorted
        
        return sorted(modules, key=lambda m: m.order)
    
    def generate_framework(
        self,
        collection: ConceptCollection,
        name: str = "Framework Chimica Organica L-13",
        n_clusters: Optional[int] = None,
        auto_optimize_clusters: bool = True,
        use_llm: bool = True
    ) -> Framework:
        """
        Pipeline completa di generazione framework.
        
        Args:
            collection: Collezione di concetti estratti
            name: Nome del framework
            n_clusters: Numero di cluster (se None, auto-ottimizza)
            auto_optimize_clusters: Se cercare automaticamente n ottimale
            use_llm: Se usare LLM per labeling e ordinamento
            
        Returns:
            Framework generato
        """
        print(f"\n{'='*60}")
        print(f"GENERAZIONE FRAMEWORK: {name}")
        print(f"{'='*60}\n")
        
        # 1. Costruisci matrice similarità
        self.build_similarity_matrix(collection)
        
        # 2. Determina numero cluster
        if n_clusters is None and auto_optimize_clusters:
            n_clusters = self.find_optimal_n_clusters()
        elif n_clusters is None:
            n_clusters = TARGET_N_MODULES
        
        # 3. Clustering
        labels = self.perform_clustering(n_clusters)
        
        # 4. Valutazione
        metrics = self.evaluate_clustering(labels)
        print(f"\nMetriche clustering:")
        print(f"  - Silhouette medio: {metrics.get('avg_silhouette', 0):.3f}")
        print(f"  - Dimensioni cluster: {metrics.get('cluster_sizes', {})}")
        
        # 5. Costruisci moduli
        modules = self.build_modules(collection, labels, use_llm_labeling=use_llm)
        
        # 6. Ordina moduli
        modules = self.order_modules(modules, use_llm=use_llm)
        
        # 7. Costruisci framework
        framework = Framework(
            id=self._generate_id("fw", name),
            name=name,
            discipline="Chimica Organica",
            degree_class="L-13",
            n_syllabus_analyzed=collection.total_syllabus_analyzed
        )
        
        for module in modules:
            # Assegna silhouette score al modulo
            cluster_silhouettes = metrics.get("silhouette_per_cluster", {})
            module.silhouette_score = cluster_silhouettes.get(module.cluster_id, 0.0)
            
            framework.add_module(module)
        
        # Normalizza pesi
        framework.normalize_weights()
        
        print(f"\n{'='*60}")
        print(f"FRAMEWORK GENERATO")
        print(f"{'='*60}")
        print(f"  - Moduli: {framework.n_modules}")
        print(f"  - Concetti totali: {framework.total_concepts}")
        print(f"  - Moduli CORE: {framework.n_core_modules}")
        print(f"  - Moduli COMUNE: {framework.n_comune_modules}")
        print(f"  - Moduli SPECIFICO: {framework.n_specifico_modules}")
        print(f"  - Silhouette medio: {framework.avg_silhouette_score:.3f}")
        
        return framework
